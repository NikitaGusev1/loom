# DAG UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the v1 localhost DAG executor UI: read-only project/DAG monitoring plus a local todo inbox.

**Architecture:** Add a small dependency-free UI subsystem under `src/ui/`. Node's built-in HTTP server serves static HTML/CSS/JS and JSON API routes. Server-side modules discover DAG files, project executor state from DAG/event/state artifacts, scan slots, and persist only `.dag-executor/ui-inbox.json`.

**Tech Stack:** TypeScript, Node 22 built-ins (`node:http`, `node:fs`, `node:path`, `node:test`), existing DAG validation/event/state/slot helpers, no frontend framework for v1.

```ts
import { existsSync, mkdirSync, readFileSync, renameSync, writeFileSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { randomUUID } from 'node:crypto';
import type { UiInboxFile, UiInboxItem, UiInboxStatus } from './types.js';

export class InboxParseError extends Error {
  path: string;

  constructor(path: string, cause: unknown) {
    super(`Failed to parse inbox JSON at ${path}: ${String(cause)}`);
    this.name = 'InboxParseError';
    this.path = path;
    this.cause = cause;
  }
}

interface ClockOptions {
  now?: () => string;
}

interface CreateInboxOptions extends ClockOptions {
  id?: () => string;
}

interface UpdateInboxOptions extends ClockOptions {
  linkedDagPath?: string | null;
  linkedTaskId?: string | null;
}

export const getInboxPath = (projectPath: string): string => join(projectPath, '.dag-executor', 'ui-inbox.json');

const nowIso = (options?: ClockOptions): string => options?.now?.() ?? new Date().toISOString();

const nextId = (options?: CreateInboxOptions): string => options?.id?.() ?? `ui-todo-${randomUUID()}`;

const assertInboxFile = (value: unknown): UiInboxFile => {
  if (!value || typeof value !== 'object') throw new Error('expected object');
  const file = value as { schemaVersion?: unknown; items?: unknown };
  if (file.schemaVersion !== 1) throw new Error('expected schemaVersion 1');
  if (!Array.isArray(file.items)) throw new Error('expected items array');
  for (const item of file.items) {
    if (!item || typeof item !== 'object') throw new Error('expected inbox item object');
    const candidate = item as Partial<UiInboxItem>;
    if (typeof candidate.id !== 'string') throw new Error('expected item.id string');
    if (typeof candidate.body !== 'string') throw new Error('expected item.body string');
    if (candidate.status !== 'open' && candidate.status !== 'converted' && candidate.status !== 'archived') {
      throw new Error('expected item.status open, converted, or archived');
    }
  }
  return file as UiInboxFile;
};

const writeInbox = (path: string, file: UiInboxFile): void => {
  mkdirSync(dirname(path), { recursive: true });
  const tmp = `${path}.tmp`;
  writeFileSync(tmp, JSON.stringify(file, null, 2) + '\n', 'utf-8');
  renameSync(tmp, path);
};

export const readInbox = (projectPath: string): UiInboxFile => {
  const path = getInboxPath(projectPath);
  if (!existsSync(path)) return { schemaVersion: 1, items: [] };
  try {
    return assertInboxFile(JSON.parse(readFileSync(path, 'utf-8')));
  } catch (cause) {
    throw new InboxParseError(path, cause);
  }
};

export const createInboxItem = (
  projectPath: string,
  body: string,
  options?: CreateInboxOptions,
): UiInboxItem => {
  const trimmed = body.trim();
  if (!trimmed) throw new Error('Inbox item body is required');

  const file = readInbox(projectPath);
  const ts = nowIso(options);
  const item: UiInboxItem = {
    id: nextId(options),
    createdAt: ts,
    updatedAt: ts,
    body: trimmed,
    status: 'open',
    source: 'ui',
    linkedDagPath: null,
    linkedTaskId: null,
  };
  writeInbox(getInboxPath(projectPath), { ...file, items: [...file.items, item] });
  return item;
};

export const updateInboxItemStatus = (
  projectPath: string,
  id: string,
  status: UiInboxStatus,
  options?: UpdateInboxOptions,
): UiInboxItem => {
  const file = readInbox(projectPath);
  const existing = file.items.find((item) => item.id === id);
  if (!existing) throw new Error(`Inbox item "${id}" not found`);

  const updated: UiInboxItem = {
    ...existing,
    status,
    updatedAt: nowIso(options),
    linkedDagPath: options?.linkedDagPath ?? existing.linkedDagPath,
    linkedTaskId: options?.linkedTaskId ?? existing.linkedTaskId,
  };
  writeInbox(getInboxPath(projectPath), {
    ...file,
    items: file.items.map((item) => (item.id === id ? updated : item)),
  });
  return updated;
};

export const archiveInboxItem = (projectPath: string, id: string, options?: ClockOptions): UiInboxItem =>
  updateInboxItemStatus(projectPath, id, 'archived', options);
```

```ts
import { existsSync, readdirSync, readFileSync, statSync } from 'node:fs';
import { join, relative, sep } from 'node:path';
import { validateDAG } from '../validation.js';
import { getEventLogPath } from '../events/log.js';
import { getStatePath } from '../state.js';
import type { UiDagListItem, UiDagStatus, UiError } from './types.js';

export interface DiscoverDagsResult {
  projectPath: string;
  dags: UiDagListItem[];
  errors: UiError[];
}

const IGNORED_DIRS = new Set([
  '.git',
  '.worktrees',
  'node_modules',
  'build',
  '.dag-executor',
  '.loom-supervisor',
  '.superpowers',
]);

const toMessage = (err: unknown): string => (err instanceof Error ? err.message : String(err));

const isCandidateDagPath = (path: string): boolean =>
  path.endsWith('.json') &&
  !path.endsWith('.state.json') &&
  !path.endsWith('.supervisor.json') &&
  !path.endsWith('package.json') &&
  !path.endsWith('package-lock.json') &&
  !path.endsWith('tsconfig.json');

const walkJson = (dir: string, out: string[]): void => {
  for (const entry of readdirSync(dir, { withFileTypes: true })) {
    const path = join(dir, entry.name);
    if (entry.isDirectory()) {
      if (!IGNORED_DIRS.has(entry.name)) walkJson(path, out);
      continue;
    }
    if (entry.isFile() && isCandidateDagPath(path)) out.push(path);
  }
};

const isoMtime = (path: string): string | null => {
  try {
    return statSync(path).mtime.toISOString();
  } catch {
    return null;
  }
};

const maxIso = (values: Array<string | null>): string | null => {
  const times = values.filter((value): value is string => value !== null).sort();
  return times[times.length - 1] ?? null;
};

const stateCounts = (
  statePath: string,
): { status: UiDagStatus; completedTasks: number; failedTasks: number; runningTasks: number } | null => {
  if (!existsSync(statePath)) return null;
  try {
    const parsed = JSON.parse(readFileSync(statePath, 'utf-8')) as {
      status?: string;
      tasks?: Record<string, { status?: string }>;
    };
    const tasks = Object.values(parsed.tasks ?? {});
    return {
      status:
        parsed.status === 'running' ||
        parsed.status === 'paused' ||
        parsed.status === 'completed' ||
        parsed.status === 'failed'
          ? parsed.status
          : 'not-started',
      completedTasks: tasks.filter((task) => task.status === 'completed').length,
      failedTasks: tasks.filter((task) => task.status === 'failed').length,
      runningTasks: tasks.filter((task) => task.status === 'in-progress').length,
    };
  } catch {
    return null;
  }
};

export const discoverDags = (projectPath: string): DiscoverDagsResult => {
  const errors: UiError[] = [];
  const candidates: string[] = [];

  try {
    walkJson(projectPath, candidates);
  } catch (err) {
    return {
      projectPath,
      dags: [],
      errors: [{ scope: 'project', path: projectPath, message: toMessage(err) }],
    };
  }

  const dags = candidates
    .sort()
    .map((path): UiDagListItem => {
      const eventLogPath = getEventLogPath(path);
      const statePath = getStatePath(path);
      const base = {
        path,
        relativePath: relative(projectPath, path).split(sep).join('/'),
        eventLogPath,
        statePath,
        updatedAt: maxIso([isoMtime(path), isoMtime(eventLogPath), isoMtime(statePath)]),
      };

      try {
        const dag = validateDAG(JSON.parse(readFileSync(path, 'utf-8')));
        const counts = stateCounts(statePath);
        return {
          ...base,
          id: dag.meta.id,
          title: dag.meta.source.title,
          status: counts?.status ?? 'not-started',
          totalTasks: dag.tasks.length,
          completedTasks: counts?.completedTasks ?? 0,
          failedTasks: counts?.failedTasks ?? 0,
          runningTasks: counts?.runningTasks ?? 0,
          errors: [],
        };
      } catch (err) {
        return {
          ...base,
          id: base.relativePath,
          title: base.relativePath,
          status: 'invalid',
          totalTasks: 0,
          completedTasks: 0,
          failedTasks: 0,
          runningTasks: 0,
          errors: [{ scope: 'dag', path, message: toMessage(err) }],
        };
      }
    });

  return { projectPath, dags, errors };
};
```

import { existsSync, readFileSync } from 'node:fs';
import { relative, sep } from 'node:path';
import { validateDAG } from '../validation.js';
import { getEventLogPath, readEvents } from '../events/log.js';
import { getStatePath } from '../state.js';
import { replay } from '../state/reducer.js';
import { analyzeBacklogReadiness } from '../backlog-readiness.js';
import { decide } from '../policy/retry.js';
import type { DAGTask, DecompositionDAG, ExecutionState, ExecutorConfig, TaskStatus } from '../types.js';
import type { DagEvent } from '../events/types.js';
import type { UiBacklogGroups, UiDagDetail, UiDagStatus, UiError, UiTaskDetail, UiTaskSummary } from './types.js';

const toMessage = (err: unknown): string => (err instanceof Error ? err.message : String(err));

const displayConfig = (dagPath: string): ExecutorConfig => ({
  provider: 'codex',
  defaultModel: 'default',
  heavyModel: 'heavy',
  maxTurns: 1,
  permissionMode: 'bypassPermissions',
  dagPath,
  dryRun: false,
});

const readDag = (dagPath: string): DecompositionDAG => validateDAG(JSON.parse(readFileSync(dagPath, 'utf-8')));

const readStateCache = (statePath: string): ExecutionState | null => {
  if (!existsSync(statePath)) return null;
  try {
    const parsed = JSON.parse(readFileSync(statePath, 'utf-8')) as ExecutionState & { derived?: true };
    if (!parsed || typeof parsed !== 'object' || !parsed.tasks || !parsed.layers) return null;
    return parsed;
  } catch {
    return null;
  }
};

const statusFromState = (state: ExecutionState | null, errors: UiError[]): UiDagStatus => {
  if (errors.some((error) => error.scope === 'dag')) return 'invalid';
  if (!state) return 'not-started';
  return state.status;
};

const taskLayerMap = (dag: DecompositionDAG): Map<string, number> => {
  const layers = new Map<string, number>();
  for (const layer of dag.executionLayers) {
    for (const taskId of layer.taskIds) layers.set(taskId, layer.layer);
  }
  return layers;
};

const progressFor = (dag: DecompositionDAG, state: ExecutionState | null) => {
  const values = dag.tasks.map((task) => state?.tasks[task.id]?.status ?? 'pending');
  const completedTasks = values.filter((status) => status === 'completed').length;
  const failedTasks = values.filter((status) => status === 'failed').length;
  const runningTasks = values.filter((status) => status === 'in-progress').length;
  const totalTasks = dag.tasks.length;
  return {
    totalTasks,
    completedTasks,
    failedTasks,
    runningTasks,
    percent: totalTasks === 0 ? 0 : Math.round((completedTasks / totalTasks) * 100),
  };
};

const summaryFor = (
  task: DAGTask,
  state: ExecutionState | null,
  layers: ReadonlyMap<string, number>,
  status?: TaskStatus | 'blocked',
  blockedBy: string[] = [],
  rootFailures: string[] = [],
): UiTaskSummary => ({
  id: task.id,
  name: task.name,
  status: status ?? state?.tasks[task.id]?.status ?? 'pending',
  layer: layers.get(task.id) ?? 0,
  dependencies: task.dependencies,
  blockedBy,
  rootFailures,
});

const backlogFor = (dag: DecompositionDAG, state: ExecutionState | null, events: DagEvent[]): UiBacklogGroups => {
  const layers = taskLayerMap(dag);
  const empty: UiBacklogGroups = { ready: [], running: [], blocked: [], failed: [], done: [], pending: [] };
  if (!state) {
    return {
      ...empty,
      pending: dag.tasks.map((task) => summaryFor(task, state, layers)),
    };
  }

  const decisions = new Map(dag.tasks.map((task) => [task.id, decide(events, task.id)]));
  const readiness = analyzeBacklogReadiness(dag, state, decisions);
  const blockedById = new Map(readiness.blocked.map((blocked) => [blocked.taskId, blocked]));
  const readyIds = new Set(readiness.ready.map((task) => task.id));

  for (const task of dag.tasks) {
    const taskState = state.tasks[task.id]?.status ?? 'pending';
    const blocked = blockedById.get(task.id);
    if (blocked) {
      empty.blocked.push(summaryFor(task, state, layers, 'blocked', blocked.blockedBy, blocked.rootFailures));
    } else if (taskState === 'completed') {
      empty.done.push(summaryFor(task, state, layers));
    } else if (taskState === 'failed') {
      empty.failed.push(summaryFor(task, state, layers));
    } else if (taskState === 'in-progress') {
      empty.running.push(summaryFor(task, state, layers));
    } else if (readyIds.has(task.id)) {
      empty.ready.push(summaryFor(task, state, layers));
    } else {
      empty.pending.push(summaryFor(task, state, layers));
    }
  }
  return empty;
};

export const loadDagDetail = (projectPath: string, dagPath: string): UiDagDetail => {
  const errors: UiError[] = [];
  const dag = readDag(dagPath);
  const eventLogPath = getEventLogPath(dagPath);
  const statePath = getStatePath(dagPath);

  let events: DagEvent[] = [];
  try {
    events = readEvents(eventLogPath);
  } catch (err) {
    errors.push({ scope: 'events', path: eventLogPath, message: toMessage(err) });
  }

  const cached = readStateCache(statePath);
  const state = cached ?? (events.length > 0 ? replay(events, dag, displayConfig(dagPath)) : null);

  return {
    dag,
    dagPath,
    relativePath: relative(projectPath, dagPath).split(sep).join('/'),
    state,
    status: statusFromState(state, errors),
    progress: progressFor(dag, state),
    backlog: backlogFor(dag, state, events),
    blockedTasks: state?.blockedTasks ?? [],
    recentEvents: events.slice(-50).reverse(),
    errors,
  };
};

export const loadTaskDetail = (projectPath: string, dagPath: string, taskId: string): UiTaskDetail => {
  const detail = loadDagDetail(projectPath, dagPath);
  const task = detail.dag.tasks.find((candidate) => candidate.id === taskId);
  if (!task) throw new Error(`Task "${taskId}" not found`);
  const layers = taskLayerMap(detail.dag);
  const state = detail.state?.tasks[taskId];
  const events = detail.recentEvents.filter((event) => 'taskId' in event && event.taskId === taskId).reverse();

  return {
    task: summaryFor(task, detail.state, layers),
    description: task.description,
    files: task.context.files,
    worktreePath: state?.worktreePath ?? null,
    worktreeBranch: state?.worktreeBranch ?? null,
    filesModified: state?.filesModified ?? [],
    usage: state?.usage,
    error: state?.error ?? null,
    events,
  };
};
```

import { createServer, type IncomingMessage, type Server, type ServerResponse } from 'node:http';
import { resolve } from 'node:path';
import { DEFAULT_SLOT_DIR } from '../semaphore.js';
import { MAX_GLOBAL_SLOTS } from '../consts.js';
import { scanSlots } from '../slots-status.js';
import { discoverDags } from './discovery.js';
import { archiveInboxItem, createInboxItem, readInbox, updateInboxItemStatus } from './inbox.js';
import { loadDagDetail, loadTaskDetail } from './projection.js';
import type { UiError } from './types.js';

export interface UiServerOptions {
  projectPath: string;
  slotDir?: string;
  maxSlots?: number;
}

const json = (res: ServerResponse, status: number, body: unknown): void => {
  res.writeHead(status, {
    'content-type': 'application/json; charset=utf-8',
    'cache-control': 'no-store',
  });
  res.end(JSON.stringify(body, null, 2));
};

const text = (res: ServerResponse, status: number, body: string, contentType = 'text/plain; charset=utf-8'): void => {
  res.writeHead(status, { 'content-type': contentType, 'cache-control': 'no-store' });
  res.end(body);
};

const errorBody = (scope: UiError['scope'], err: unknown, path?: string): { error: UiError } => ({
  error: { scope, path, message: err instanceof Error ? err.message : String(err) },
});

const readBody = async (req: IncomingMessage): Promise<unknown> => {
  const chunks: Buffer[] = [];
  for await (const chunk of req) chunks.push(Buffer.isBuffer(chunk) ? chunk : Buffer.from(chunk));
  if (chunks.length === 0) return {};
  return JSON.parse(Buffer.concat(chunks).toString('utf-8'));
};

export const createUiServer = (options: UiServerOptions): Server => {
  const projectPath = resolve(options.projectPath);
  const slotDir = options.slotDir ?? DEFAULT_SLOT_DIR;
  const maxSlots = options.maxSlots ?? MAX_GLOBAL_SLOTS;

  return createServer(async (req, res) => {
    const url = new URL(req.url ?? '/', 'http://localhost');

    try {
      if (req.method === 'GET' && url.pathname === '/api/project') {
        const discovered = discoverDags(projectPath);
        json(res, 200, {
          projectPath,
          dagCount: discovered.dags.length,
          slots: scanSlots(slotDir, maxSlots),
          errors: discovered.errors,
        });
        return;
      }

      if (req.method === 'GET' && url.pathname === '/api/dags') {
        json(res, 200, discoverDags(projectPath));
        return;
      }

      if (req.method === 'GET' && url.pathname === '/api/dags/detail') {
        const dagPath = url.searchParams.get('path');
        if (!dagPath) {
          json(res, 400, { error: { scope: 'dag', message: 'Missing path query parameter' } });
          return;
        }
        json(res, 200, loadDagDetail(projectPath, dagPath));
        return;
      }

      if (req.method === 'GET' && url.pathname === '/api/tasks/detail') {
        const dagPath = url.searchParams.get('dagPath');
        const taskId = url.searchParams.get('taskId');
        if (!dagPath || !taskId) {
          json(res, 400, { error: { scope: 'dag', message: 'Missing dagPath or taskId query parameter' } });
          return;
        }
        json(res, 200, loadTaskDetail(projectPath, dagPath, taskId));
        return;
      }

      if (req.method === 'GET' && url.pathname === '/api/inbox') {
        json(res, 200, readInbox(projectPath));
        return;
      }

      if (req.method === 'POST' && url.pathname === '/api/inbox') {
        const body = (await readBody(req)) as { body?: unknown };
        if (typeof body.body !== 'string') {
          json(res, 400, { error: { scope: 'inbox', message: 'Request body must include string body' } });
          return;
        }
        json(res, 201, { item: createInboxItem(projectPath, body.body) });
        return;
      }

      const archiveMatch = url.pathname.match(/^\/api\/inbox\/([^/]+)\/archive$/);
      if (req.method === 'POST' && archiveMatch) {
        json(res, 200, { item: archiveInboxItem(projectPath, decodeURIComponent(archiveMatch[1])) });
        return;
      }

      const updateMatch = url.pathname.match(/^\/api\/inbox\/([^/]+)$/);
      if (req.method === 'PATCH' && updateMatch) {
        const body = (await readBody(req)) as { status?: unknown; linkedDagPath?: unknown; linkedTaskId?: unknown };
        if (body.status !== 'open' && body.status !== 'converted' && body.status !== 'archived') {
          json(res, 400, { error: { scope: 'inbox', message: 'status must be open, converted, or archived' } });
          return;
        }
        json(res, 200, {
          item: updateInboxItemStatus(projectPath, decodeURIComponent(updateMatch[1]), body.status, {
            linkedDagPath: typeof body.linkedDagPath === 'string' ? body.linkedDagPath : undefined,
            linkedTaskId: typeof body.linkedTaskId === 'string' ? body.linkedTaskId : undefined,
          }),
        });
        return;
      }

      text(res, 404, 'Not found');
    } catch (err) {
      json(res, 500, errorBody('project', err, projectPath));
    }
  });
};
```

### Local UI

Start the read-only localhost UI for a project or repo:

```sh
dag-executor --ui --project=/path/to/project
```

The command prints the local URL, for example:

```text
DAG UI listening on http://127.0.0.1:51234
```

The v1 UI discovers DAG JSON files under the project, reads `.events.jsonl` and `.state.json` artifacts for display, scans machine-wide executor slots, and persists local inbox items to `.dag-executor/ui-inbox.json`.

V1 is intentionally read-only for executor state. It does not start, stop, resume, rerun, edit DAG files, append events, mutate `.state.json`, change queue priority, or run git commands from the browser.
